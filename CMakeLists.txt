cmake_minimum_required(VERSION 3.25)
project(arkpets_linux VERSION 0.9.6 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Network)
qt_standard_project_setup(REQUIRES 6.8)
include(GNUInstallDirs)
include(CTest)
# Source integration recommended by sudoevolve/material-components-qml.
add_subdirectory(vendor/material-components-qml/src/Core EXCLUDE_FROM_ALL)
# NO_PLUGIN modules need their registrations and resources retained at link time.
add_library(arkpets-material INTERFACE)
target_link_libraries(arkpets-material INTERFACE Qt6::Quick "$<LINK_LIBRARY:WHOLE_ARCHIVE,md3Core>")
set(ARKPETS_UI_FILES qml/Main.qml qml/PanelSettings.qml qml/ModelSettings.qml qml/MultiplierField.qml qml/MdText.qml qml/NumberField.qml qml/WindowResizeHandles.qml resources/arkpets.svg)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET gtk4>=4.12 gtk4-layer-shell-0 epoxy)
file(GLOB SPINE_SOURCES CONFIGURE_DEPENDS vendor/spine-c/src/spine/*.c)
add_library(spine STATIC ${SPINE_SOURCES})
set_target_properties(spine PROPERTIES AUTOMOC OFF)
target_include_directories(spine PUBLIC vendor/spine-c/include)
target_link_libraries(spine PUBLIC m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(spine PRIVATE "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.")
endif()
add_executable(arkpets-renderer src/main.cpp)
set_target_properties(arkpets-renderer PROPERTIES AUTOMOC OFF)
target_link_libraries(arkpets-renderer PRIVATE spine PkgConfig::NATIVE)
target_compile_options(arkpets-renderer PRIVATE -Wall -Wextra -Wpedantic)
add_library(arkpets-backend STATIC src/gui/catalog.cpp src/gui/catalog.h src/gui/controller.cpp src/gui/controller.h
    src/gui/panelsettings.cpp src/gui/panelsettings.h src/gui/listwheelhandler.cpp src/gui/listwheelhandler.h
    src/gui/windowtracker.cpp src/gui/windowtracker.h)
target_link_libraries(arkpets-backend PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Quick)
target_include_directories(arkpets-backend PUBLIC src/gui)
target_compile_options(arkpets-backend PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(arkpets src/gui/main.cpp)
target_link_libraries(arkpets PRIVATE arkpets-backend arkpets-material Qt6::Widgets Qt6::Quick Qt6::QuickControls2)
qt_add_resources(arkpets ui PREFIX "/ui" FILES ${ARKPETS_UI_FILES})
set_target_properties(arkpets arkpets-renderer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
install(TARGETS arkpets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS arkpets-renderer RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/arkpets)
install(FILES packaging/arkpets.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/arkpets.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/arkpets-linux)
install(FILES vendor/spine-c/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/arkpets-linux RENAME SPINE-LICENSE)
install(FILES vendor/material-components-qml/NOTICE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/arkpets-linux/material-components-qml)
install(DIRECTORY vendor/material-components-qml/LICENSES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/arkpets-linux/material-components-qml)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/arkpets-linux)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(arkpets-tests tests/backend_test.cpp)
    target_link_libraries(arkpets-tests PRIVATE arkpets-backend arkpets-material Qt6::Test Qt6::Quick Qt6::QuickControls2)
    qt_add_resources(arkpets-tests test_ui PREFIX "/ui" FILES ${ARKPETS_UI_FILES})
    target_compile_definitions(arkpets-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_SOURCE_DIR}/tests")
    set_target_properties(arkpets-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
    add_dependencies(arkpets-tests arkpets-renderer)
    add_test(NAME backend COMMAND arkpets-tests)
    set_tests_properties(backend PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_STYLE_OVERRIDE=Fusion;GSETTINGS_BACKEND=memory")
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
    find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/virtual-pointer.h" "${CMAKE_BINARY_DIR}/virtual-pointer-protocol.c"
        COMMAND ${WAYLAND_SCANNER} client-header "${CMAKE_SOURCE_DIR}/tests/protocol/wlr-virtual-pointer-unstable-v1.xml" "${CMAKE_BINARY_DIR}/virtual-pointer.h"
        COMMAND ${WAYLAND_SCANNER} private-code "${CMAKE_SOURCE_DIR}/tests/protocol/wlr-virtual-pointer-unstable-v1.xml" "${CMAKE_BINARY_DIR}/virtual-pointer-protocol.c"
        DEPENDS tests/protocol/wlr-virtual-pointer-unstable-v1.xml)
    add_executable(drag-probe tests/drag_probe.c "${CMAKE_BINARY_DIR}/virtual-pointer-protocol.c" "${CMAKE_BINARY_DIR}/virtual-pointer.h")
    set_target_properties(drag-probe PROPERTIES AUTOMOC OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
    target_include_directories(drag-probe PRIVATE "${CMAKE_BINARY_DIR}")
    target_link_libraries(drag-probe PRIVATE PkgConfig::WAYLAND)
endif()
